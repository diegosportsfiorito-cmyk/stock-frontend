# indexer.py import io import pandas as pd import numpy as np from typing import List, Dict from datetime import datetime from drive import listar_archivos_en_carpeta, descargar_archivo_por_id CARPETA_STOCK_ID = "1F0FUEMJmeHgb3ZY7XBBdacCGB3SZK4O-" # ============================================================ # 1) DETECCIÓN AUTOMÁTICA DE ENCABEZADOS # ============================================================ def detectar_encabezados(df: pd.DataFrame) -> pd.DataFrame: """ Si la primera fila contiene texto, se usa como encabezado. Si contiene números, se generan encabezados genéricos. """ primera_fila = df.iloc[0] # Si la primera fila tiene al menos 1 texto → es encabezado if any(isinstance(x, str) for x in primera_fila): df.columns = [str(c).strip() for c in primera_fila] df = df[1:].reset_index(drop=True) else: # Generar encabezados genéricos df.columns = [f"col_{i}" for i in range(len(df.columns))] return df # ============================================================ # 2) NORMALIZACIÓN DE COLUMNAS # ============================================================ COLUMNAS_TEXTO = ["articulo", "descripcion", "color", "rubro", "base", "nombre"] COLUMNAS_TALLE = ["talle", "size", "tamaño"] COLUMNAS_PRECIO = ["precio", "lista1", "lista0", "valor", "valorizado"] COLUMNAS_STOCK = ["stock", "cantidad", "cant"] def normalizar_columnas(df: pd.DataFrame) -> pd.DataFrame: df = df.copy() df.columns = [str(c).strip().lower() for c in df.columns] return df # ============================================================ # 3) DETECCIÓN AUTOMÁTICA DE COLUMNAS CLAVE # ============================================================ def detectar_columnas(df: pd.DataFrame) -> Dict[str, str]: columnas = { "articulo": None, "descripcion": None, "talle": None, "precio": None, "stock": None } for col in df.columns: col_lower = col.lower() # ARTÍCULO if any(k in col_lower for k in ["art", "cod", "item"]): columnas["articulo"] = col # DESCRIPCIÓN if any(k in col_lower for k in ["desc", "nombre", "color"]): columnas["descripcion"] = col # TALLE if any(k in col_lower for k in ["talle", "size", "tam"]): columnas["talle"] = col # PRECIO if any(k in col_lower for k in ["precio", "lista", "valor"]): columnas["precio"] = col # STOCK if any(k in col_lower for k in ["stock", "cant"]): columnas["stock"] = col return columnas # ============================================================ # 4) LIMPIEZA DE PRECIOS (coma, punto o sin separador) # ============================================================ def limpiar_precio(valor): if isinstance(valor, str): valor = valor.replace(".", "").replace(",", ".") try: return float(valor) except: return np.nan # ============================================================ # 5) BÚSQUEDA SEMÁNTICA HÍBRIDA # ============================================================ def buscar_semantico(df: pd.DataFrame, consulta: str, columnas: Dict[str, str]) -> pd.DataFrame: consulta = consulta.lower().strip() # Detectar si la consulta contiene un número numeros = "".join([c for c in consulta if c.isdigit() or c == "."]) # -------------------------------------------------------- # MODO 1: BÚSQUEDA POR CÓDIGO EXACTO # -------------------------------------------------------- if numeros and len(numeros) >= 4: coincidencias = df.apply(lambda row: row.astype(str).str.contains(numeros, case=False, regex=False).any(), axis=1) return df[coincidencias] # -------------------------------------------------------- # MODO 2: BÚSQUEDA POR TEXTO PARCIAL # -------------------------------------------------------- coincidencias = df.apply(lambda row: row.astype(str).str.lower().str.contains(consulta).any(), axis=1) return df[coincidencias] # ============================================================ # 6) GENERAR CONTEXTO PARA EL MODELO # ============================================================ def obtener_contexto_para_pregunta(pregunta: str) -> List[Dict]: archivos = listar_archivos_en_carpeta(CARPETA_STOCK_ID) contextos = [] for archivo in archivos: nombre = archivo["name"] file_id = archivo["id"] mod_time = archivo.get("modifiedTime") if not (nombre.lower().endswith(".xlsx") or nombre.lower().endswith(".xls")): continue file_bytes = descargar_archivo_por_id(file_id) excel_file = io.BytesIO(file_bytes) try: xls = pd.ExcelFile(excel_file) except: continue filas_encontradas = [] for sheet in xls.sheet_names: try: df = xls.parse(sheet, header=None) df = detectar_encabezados(df) df = normalizar_columnas(df) columnas = detectar_columnas(df) df["precio_limpio"] = df[columnas["precio"]].apply(limpiar_precio) if columnas["precio"] else np.nan filtrado = buscar_semantico(df, pregunta, columnas) if not filtrado.empty: filas_encontradas.append(filtrado) except Exception: pass if not filas_encontradas: continue df_final = pd.concat(filas_encontradas, ignore_index=True) fecha_str = None if mod_time: try: dt = datetime.fromisoformat(mod_time.replace("Z", "+00:00")) fecha_str = dt.strftime("%Y-%m-%d") except: fecha_str = mod_time contextos.append({ "archivo": nombre, "fecha": fecha_str or "Fecha desconocida", "contenido": df_final.to_string(index=False) }) return contextos
